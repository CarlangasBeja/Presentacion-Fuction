🚗 SISTEMA DE VENTAS DE AUTOS

Simulación de Functions y consultas SQL

🗄️ Crear tablas

CREATE TABLE CLIENTE(
    id_cliente INT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL,
    ciudad VARCHAR(50) NOT NULL
);

CREATE TABLE AUTO(
    id_auto INT PRIMARY KEY,
    marca VARCHAR(50) NOT NULL,
    modelo VARCHAR(50) NOT NULL,
    precio DECIMAL(10,2) CHECK (precio > 0)
);

CREATE TABLE VENTA(
    id_venta INT PRIMARY KEY,
    id_cliente INT,
    FOREIGN KEY (id_cliente) REFERENCES CLIENTE(id_cliente)
);

CREATE TABLE DETALLE_VENTA(
    id_detalle INT PRIMARY KEY,
    id_venta INT,
    id_auto INT,
    cantidad INT CHECK (cantidad > 0),
    FOREIGN KEY (id_venta) REFERENCES VENTA(id_venta),
    FOREIGN KEY (id_auto) REFERENCES AUTO(id_auto)
);

⚙️ Function: descuento

CREATE FUNCTION descuento(precio DECIMAL)
RETURNS DECIMAL
BEGIN
RETURN precio - (precio * 0.10);
END;

🧪 Ejercicio 1

SELECT marca, descuento(precio) FROM AUTO;

⚙️ Function: total_venta

CREATE FUNCTION total_venta(precio DECIMAL, cantidad INT)
RETURNS DECIMAL
BEGIN
RETURN precio * cantidad;
END;

🧪 Ejercicio 2

SELECT marca, total_venta(precio,cantidad)
FROM AUTO a JOIN DETALLE_VENTA dv
ON a.id_auto=dv.id_auto;
